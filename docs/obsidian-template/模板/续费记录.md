<%*
/*
 * 记一笔续费（Templater 模板）。在某张卡片的笔记里运行「Templater: 插入模板」并选它：
 * 弹出选择框，把一行记录加到这篇笔记的续费表格末尾，光标处不会插入任何东西。
 * 「有效期延至」默认 = 当前有效期 + 续费周期，可以改。
 */
const OPTIONS_PATH = '模板/下拉选项.md'
const file = tp.config.target_file
const fm = app.metadataCache.getFileCache(file)?.frontmatter ?? {}

// 下拉选项
const optionsFile = app.vault.getAbstractFileByPath(OPTIONS_PATH)
const groups = {}
if (optionsFile) {
  let current = null
  for (const line of (await app.vault.read(optionsFile)).split('\n')) {
    const heading = line.match(/^##\s+(.+?)\s*$/)
    if (heading) {
      current = heading[1]
      groups[current] = []
      continue
    }
    const item = line.match(/^-\s+(.+?)\s*$/)
    if (current && item)
      groups[current].push(item[1].split('|').map(s => s.trim()))
  }
}
const group = name => (groups[name] ?? []).map(([first]) => first)

async function ask(prompt, defaultValue = '') {
  const value = await tp.system.prompt(prompt, defaultValue, false)
  return (value ?? '').trim()
}

// 找到正文里第一张有「日期」列的表格
const text = await app.vault.read(file)
const lines = text.split('\n')
const isRow = line => /^\s*\|.*\|\s*$/.test(line)
const cells = line => line.trim().replace(/^\||\|$/g, '').split(/(?<!\\)\|/).map(c => c.trim())
let header = -1
for (let i = 0; i + 1 < lines.length; i++) {
  if (isRow(lines[i]) && isRow(lines[i + 1]) && cells(lines[i]).some(c => c.replace(/\s/g, '') === '日期') && /^[\s|:-]+$/.test(lines[i + 1])) {
    header = i
    break
  }
}

if (header === -1) {
  new tp.obsidian.Notice('这篇笔记里没有带「日期」列的续费表格')
}
else {
  const columns = cells(lines[header]).map(c => c.replace(/\s/g, ''))
  let end = header + 2
  while (end < lines.length && isRow(lines[end]))
    end++

  // 当前有效期：消费日期最近的那一行的「有效期延至」
  const col = name => columns.indexOf(name)
  let latest = null
  for (const line of lines.slice(header + 2, end)) {
    const row = cells(line)
    const date = row[col('日期')] ?? ''
    const ext = row[col('有效期延至')] ?? ''
    if (/^\d{4}-\d{2}-\d{2}$/.test(date) && /^\d{4}-\d{2}-\d{2}$/.test(ext) && (!latest || date > latest.date || (date === latest.date && ext > latest.ext)))
      latest = { date, ext }
  }

  const cycle = String(fm['续费周期'] ?? '')
  const cycleMatch = cycle.match(/(\d+(?:\.\d+)?)/)
  let cycleDays = cycleMatch ? Number(cycleMatch[1]) : 0
  if (/年/.test(cycle))
    cycleDays *= 365
  else if (/月/.test(cycle))
    cycleDays *= 30
  const cardCurrency = String(fm['币种'] ?? 'CNY') || 'CNY'
  const today = tp.date.now('YYYY-MM-DD')
  const base = latest && latest.ext > today ? latest.ext : today

  const date = await ask('日期', today)
  const type = await tp.system.suggester(group('记录类型'), group('记录类型'), false, '记录类型') ?? '续费/保号'
  const amount = await ask(`金额（${cardCurrency}）`, fm['续费金额'] != null ? String(fm['续费金额']) : '')
  const currencies = [...new Set([cardCurrency, ...group('币种')])]
  const currency = await tp.system.suggester(currencies, currencies, false, '币种') ?? cardCurrency
  const rate = currency === 'CNY' ? '' : await ask(`当时汇率（1 ${currency} = ? 人民币，不知道可留空）`)
  const extendsTo = await ask('有效期延至', cycleDays && type !== '额外充值' ? tp.date.now('YYYY-MM-DD', Math.round(cycleDays), base, 'YYYY-MM-DD') : '')
  const note = await ask('备注（可留空）')

  const values = { 日期: date, 类型: type, 金额: amount, 币种: currency, 汇率: rate, 有效期延至: extendsTo, 备注: note }
  const row = `| ${columns.map(c => (values[c] ?? '').replace(/\|/g, '\\|')).join(' | ')} |`
  await app.vault.process(file, (content) => {
    const current = content.split('\n')
    current.splice(end, 0, row)
    return current.join('\n')
  })
  new tp.obsidian.Notice(`已记一笔：${date} ${type}`)
}
-%>
